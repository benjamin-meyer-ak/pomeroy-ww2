5  Local OCR

5.1 Method

Every page is run through Tesseract in its automatic page-segmentation mode (psm 3) on a grayscale image. For each page the script saves the full text and a word table with confidence and bounding box. Results are cached one file per page, so an interrupted run resumes and nothing is recomputed unnecessarily.

Show code 
# Page pipeline: PDF page -> cropped 300 DPI PNG -> tesseract OCR -> word table
# and page text, cached per page so a run can be stopped and resumed.
#   data/pages/<page_id>.png        prepared page image
#   data/interim/ocr/<page_id>.rds  list(text, words, langs)
library(tidyverse)
library(pdftools)
library(magick)
library(tesseract)
source("R/utils.R")
source("R/crop_spread.R")

cfg <- read_csv("data/volume_config.csv", show_col_types = FALSE)


page_ids <- function(vol) {
  row <- filter(cfg, volume_id == vol)
  tibble(volume_id = vol, file = row$file, page = seq_len(row$n_pages),
         page_id = sprintf("%s_p%03d", vol, page),
         layout = row$layout, is_portrait = page %in% parse_pages(row$portrait_pages))
}

# Download any tesseract language data that volume_config.csv asks for and is
# not installed. Stops if a language is still missing, so a run never silently
# uses fewer languages than the config says.
ensure_languages <- function(config) {
  want <- config$ocr_langs |> str_split("\\+") |> unlist() |> unique()
  for (lang in setdiff(want, tesseract_info()$available)) {
    message("Downloading tesseract language data: ", lang)
    tesseract_download(lang, progress = FALSE)
  }
  missing <- setdiff(want, tesseract_info()$available)
  if (length(missing)) stop("Tesseract language data still missing: ", paste(missing, collapse = ", "))
  sort(want)
}

prepare_page <- function(vol, page, force = FALSE) {
  id <- sprintf("%s_p%03d", vol, page)
  out <- file.path("data/pages", paste0(id, ".png"))
  if (!force && file.exists(out)) return(out)
  row <- filter(cfg, volume_id == vol)
  pdf <- file.path("data/raw_pdf", row$file)
  img <- if (row$layout == "spread" && !(page %in% parse_pages(row$portrait_pages))) {
    suppressWarnings(crop_text_half(pdf, page))
  } else {
    tmp <- file.path("data/interim", paste0("full_", id, ".png"))
    suppressWarnings(pdf_convert(pdf, pages = page, dpi = 300, format = "png",
                                 filenames = tmp, verbose = FALSE))
    on.exit(unlink(tmp))
    image_read(tmp)
  }
  dir.create(dirname(out), showWarnings = FALSE, recursive = TRUE)
  image_write(img, out, format = "png")
  out
}

ocr_engine <- function(langs = "eng", psm = 3) {
  tesseract(langs, options = list(tessedit_pageseg_mode = as.character(psm)))
}

# A cached result is reused only if it was made with the same languages, so
# changing ocr_langs in the config reruns the affected pages.
ocr_page <- function(vol, page, langs = "eng", force = FALSE) {
  id <- sprintf("%s_p%03d", vol, page)
  cache <- file.path("data/interim/ocr", paste0(id, ".rds"))
  if (!force && file.exists(cache) && identical(readRDS(cache)$langs, langs)) {
    return(invisible(cache))
  }
  png <- prepare_page(vol, page, force = force)
  img <- image_read(png) |> image_convert(colorspace = "gray")
  raw <- image_write(img, format = "png")
  eng <- ocr_engine(langs)
  words <- ocr_data(raw, engine = eng) |>
    mutate(page_id = id, .before = 1) |>
    separate_wider_delim(bbox, ",", names = c("x0", "y0", "x1", "y1")) |>
    mutate(across(x0:y1, as.integer))
  text <- ocr(raw, engine = eng)
  dir.create(dirname(cache), showWarnings = FALSE, recursive = TRUE)
  saveRDS(list(text = text, words = words, langs = langs), cache)
  invisible(cache)
}

The whole set of 378 pages runs in parallel across worker processes. With six workers it takes about 2.6 seconds per page, roughly 20 minutes in total. Pages already in the cache are skipped unless they were made with different languages. Languages come from data/volume_config.csv, and ensure_languages() installs any that are missing before the run starts.

Show code 
# Run the page pipeline (prepare_page + ocr_page) for whole volumes in parallel.
# Results are cached per page, so a rerun only processes pages that are missing
# or were made with different languages (force = TRUE redoes everything).
# About 2.6 s/page with 6 workers. Returns the paths of all cached results.
source("R/ocr_pipeline.R")
library(parallel)

run_ocr <- function(config, pdf_files = NULL, workers = 6, force = FALSE) {
  jobs <- map(config$volume_id, \(v) {
    # Languages come straight from the config; ensure_languages() has installed them
    page_ids(v) |> mutate(langs = filter(config, volume_id == v)$ocr_langs)
  }) |> list_rbind()

  cl <- makeCluster(workers)
  on.exit(stopCluster(cl))
  clusterEvalQ(cl, source("R/ocr_pipeline.R"))
  parLapplyLB(cl, seq_len(nrow(jobs)), \(i, jobs, force) {
    ocr_page(jobs$volume_id[i], jobs$page[i], langs = jobs$langs[i], force = force)
  }, jobs = jobs, force = force) |>
    invisible()
  file.path("data/interim/ocr", paste0(jobs$page_id, ".rds"))
}

5.2 Settings that were tried

Two small experiments ran on the prepared page images, with the code below. Both are rerun by the pipeline and their results are saved in data/interim/experiments/, so the numbers in this chapter are regenerated, not typed in.

Show code 
# Tuning experiments behind the choices described in the "Local OCR" chapter.
# Both run on the prepared page images (data/pages/), so run the OCR step
# first. Results are written to data/interim/experiments/ and read by the book.
library(tidyverse)
library(magick)
library(tesseract)

# Ten pages that cover every layout: portrait singles, a TOC, spreads, a
# faded-ribbon page and a two-column page
sample_ids <- c("vol1_p001", "vol1_p044", "vol1_p070", "vol2_p001", "vol2_p040",
                "vol2_p120", "vol3_p002", "vol3_p020", "vol3_p066", "vol3_p087")

ocr_stats <- function(img, psm = 3) {
  raw <- image_write(image_convert(img, colorspace = "gray"), format = "png")
  d <- ocr_data(raw, engine = tesseract("eng", options = list(tessedit_pageseg_mode = as.character(psm))))
  tibble(words = nrow(d), mean_conf = mean(d$confidence), pct_below_80 = mean(d$confidence < 80) * 100)
}

# Page-segmentation modes 3 (automatic), 4 (single column), 6 (single block)
compare_psm <- function(ids = sample_ids, psms = c(3, 4, 6)) {
  expand_grid(page_id = ids, psm = psms) |>
    mutate(stats = map2(page_id, psm, \(id, m) ocr_stats(image_read(file.path("data/pages", paste0(id, ".png"))), m))) |>
    unnest(stats)
}

# Preprocessing variants on one page from each of three kinds of page
preprocess_variants <- list(
  baseline = identity,
  threshold = \(img) img |> image_convert(colorspace = "gray") |>
    image_threshold("white", "55%") |> image_threshold("black", "55%"),
  contrast = \(img) image_contrast(img, sharpen = 1)
)

compare_preprocessing <- function(ids = c("vol1_p010", "vol2_p040", "vol3_p020")) {
  expand_grid(page_id = ids, variant = names(preprocess_variants)) |>
    mutate(stats = map2(page_id, variant, \(id, v) {
      img <- image_read(file.path("data/pages", paste0(id, ".png")))
      ocr_stats(preprocess_variants[[v]](img))
    })) |>
    unnest(stats)
}

run_experiments <- function(ocr_files = NULL, dir = "data/interim/experiments") {
  dir.create(dir, showWarnings = FALSE, recursive = TRUE)
  paths <- c(psm = file.path(dir, "psm_comparison.csv"),
             preprocessing = file.path(dir, "preprocessing_comparison.csv"))
  write_csv(compare_psm(), paths[["psm"]])
  write_csv(compare_preprocessing(), paths[["preprocessing"]])
  unname(paths)
}

5.2.1 Page-segmentation mode

Ten pages covering every layout were run under three Tesseract page-segmentation modes: psm 3 (automatic), psm 4 (single column) and psm 6 (single block).

Show code 
psm <- read_csv("data/interim/experiments/psm_comparison.csv", show_col_types = FALSE)
psm |>
  summarise(.by = psm, mean_conf = mean(mean_conf), words_per_page = mean(words),
            pct_below_80 = mean(pct_below_80)) |>
  kable(digits = 1)
Show code 
psm |>
  select(page_id, psm, mean_conf) |>
  pivot_wider(names_from = psm, names_prefix = "psm ", values_from = mean_conf) |>
  kable(digits = 1)

On ordinary pages the three modes are within a point or two of each other. The difference appears on the two-column page (vol3_p087): psm 6 reads across both columns as one block and scores about ten points lower with more words, which is the signature of merged columns. psm 3 was used for the full pass as a reasonable default. Mean confidence is only a proxy for accuracy, so this is not a proven best.

5.2.2 Preprocessing

Hard thresholding (pure black and white) and a contrast boost were compared with the unprocessed image on one page from each of three volumes.

Show code 
read_csv("data/interim/experiments/preprocessing_comparison.csv", show_col_types = FALSE) |>
  select(page_id, variant, words, mean_conf) |>
  pivot_wider(names_from = variant, values_from = c(words, mean_conf)) |>
  kable(digits = 1)

Thresholding is not a safe improvement. It leaves the faded-ribbon page (vol1_p010) about where it was and cuts the word count, and it drops a clean page (vol2_p040) from 93 to about 33. Contrast changes little either way. No preprocessing is applied.

5.3 What is still unknown

The choice of mode, preprocessing and languages should be made by comparing OCR output with hand-typed text and computing the character error rate. No such reference pages exist yet. When they do, the comparison should cover each layout: clean portrait, spread, faded ribbon, two-column and front matter.

5.4 Languages

The first pass used English only. The 1966 journal has German, French and Dutch names, which English-only OCR tends to garble. The config asks for eng+deu+fra for volume 2, and the pipeline installs the missing models and redoes the pages whose cached result used different languages. Rerun before reviewing volume 2: saved corrections replace the OCR text for their page, so a later rerun would not improve pages already corrected.
